function SearchDrop({value,onChange,items,placeholder,renderLabel,renderSub,filterFn}){

  const [q,setQ]=useState('');

  const [open,setOpen]=useState(false);

  const ref=useRef();

  const sel=items.find(i=>i.id===value);

 

  useEffect(()=>{

    const h=e=>{if(ref.current&&!ref.current.contains(e.target)){setOpen(false);setQ('');}};

    document.addEventListener('mousedown',h);return()=>document.removeEventListener('mousedown',h);

  },[]);

 

  const visible=q?items.filter(i=>filterFn?filterFn(i,q):hits(renderLabel(i),q)):items;

 

  return React.createElement('div',{className:'sdrop-wrap',ref},

    React.createElement('div',{className:'sdrop-input-row'},

      React.createElement('input',{

        value:open?q:(sel?renderLabel(sel):''),

        placeholder:sel?renderLabel(sel):placeholder,

        onFocus:()=>{setOpen(true);setQ('');},

        onChange:e=>{setQ(e.target.value);setOpen(true);},

      }),

      sel&&!open&&React.createElement('button',{className:'sdrop-arrow',onMouseDown:e=>{e.preventDefault();onChange('');setQ('');}},,'✕'),

      React.createElement('button',{className:'sdrop-arrow',onMouseDown:e=>{e.preventDefault();setOpen(o=>!o);}},'▾'),

    ),

    open&&React.createElement('div',{className:'sdrop-list'},

      visible.length===0

        ?React.createElement('div',{className:'sdrop-empty'},'No results')

        :visible.map(i=>React.createElement('div',{key:i.id,className:`sdrop-item ${value===i.id?'hi':''}`,

            onMouseDown:e=>{e.preventDefault();onChange(i.id);setQ('');setOpen(false);}},

            React.createElement('div',{className:'sdrop-main'},renderLabel(i)),

            renderSub&&React.createElement('div',{className:'sdrop-sub'},renderSub(i)),

          ))

    )

  );

}
